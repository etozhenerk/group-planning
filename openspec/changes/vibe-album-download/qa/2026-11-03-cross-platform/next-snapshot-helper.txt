package albumexport

import (
    "archive/zip"
    "bytes"
    "io"
    "testing"
)

func TestQANextExportIncludesNewlyAddedPhoto(t *testing.T) {
    source := albumsFixture(103)
    source.gate, source.entered = make(chan struct{}), make(chan struct{})
    manager := newManager(t, source)
    first := begin(t, manager, testActor, "qa-first")
    <-source.entered
    source.add(103)
    close(source.gate)
    awaitState(t, manager, testActor, first.ID, "ready")
    read := func(id string, expected int) {
        t.Helper()
        stream, err := manager.Open(t.Context(), testActor, "meeting", id, validSession)
        if err != nil { t.Fatal(err) }
        data, err := io.ReadAll(stream)
        if closeErr := stream.Close(); closeErr != nil { t.Fatal(closeErr) }
        if err != nil { t.Fatal(err) }
        archive, err := zip.NewReader(bytes.NewReader(data), int64(len(data)))
        if err != nil { t.Fatal(err) }
        if len(archive.File) != expected { t.Fatalf("archive has %d files, want %d",len(archive.File),expected) }
    }
    read(first.ID,103)
    second := begin(t, manager, testActor,"qa-next")
    if second.Count != 104 { t.Fatalf("new snapshot count %d", second.Count) }
    awaitState(t,manager,testActor,second.ID,"ready")
    read(second.ID,104)
    t.Log("REAL_ZIP_SNAPSHOTS first=103 next=104")
}
