import assert from 'node:assert/strict';
import fs from 'node:fs';
import {Centrifuge} from 'centrifuge';
import WS from 'ws';
const base=fs.readFileSync('../endpoint.txt','utf8');
const sleep=ms=>new Promise(r=>setTimeout(r,ms));
const until=async(fn,label)=>{let t=Date.now();while(!fn()){if(Date.now()-t>10000)throw Error('timeout: '+label);await sleep(20)}};
const ctl=async(op)=>{let r=await fetch(base+'/control/'+op+'?who=js');assert(r.ok);return r.status===204?null:r.json()};
const initialCursor=(await ctl('sync')).change_cursor;
const events=[]; const subscriptions=[]; const hints=[]; let catches=0;
class CookieSocket extends WS {constructor(url,protocols){super(url,protocols,{headers:{Cookie:'spike_session=web-js',Origin:'http://spike.local'}})}}
const client=new Centrifuge(base.replace('http','ws')+'/api/v1/realtime',{websocket:CookieSocket,minReconnectDelay:800,maxReconnectDelay:800,timeout:2000});
client.on('error',x=>events.push({type:'error',code:x.error?.code}));
for(const event of ['connected','connecting','disconnected'])client.on(event,x=>events.push({type:event,code:x.code}));
const sub=client.newSubscription('meeting:js');
sub.on('subscribed',x=>{subscriptions.push({wasRecovering:x.wasRecovering,recovered:x.recovered});ctl('sync').then(()=>catches++)});
sub.on('publication',x=>{assert.deepEqual(Object.keys(x.data).sort(),['latest_change_sequence','meeting_id','schema_version','stream_epoch']);assert.equal(typeof x.data.latest_change_sequence,'string');hints.push(x.data.latest_change_sequence)});
sub.subscribe();client.connect();
await until(()=>subscriptions.length===1&&catches===1,'initial subscribed and HTTP catch-up');
assert.equal((await fetch(base+'/api/v1/realtime')).status,401);
let badOrigin=await new Promise(resolve=>{const ws=new WS(base.replace('http','ws')+'/api/v1/realtime',{headers:{Cookie:'spike_session=web-js',Origin:'http://evil.invalid'}});ws.on('unexpected-response',(_,res)=>{resolve(res.statusCode);ws.terminate()});ws.on('error',()=>{});});
assert.equal(badOrigin,403);
const outsider=new Centrifuge(base.replace('http','ws')+'/api/v1/realtime',{websocket:CookieSocket});outsider.on('error',()=>{});
const forbidden=outsider.newSubscription('meeting:other');let denied;
forbidden.on('unsubscribed',x=>denied=x.code);forbidden.on('error',()=>{});forbidden.subscribe();outsider.connect();await until(()=>denied===103,'foreign channel denied');outsider.disconnect();
await assert.rejects(client.publish('meeting:js',{text:'must not be accepted'}),e=>e.code===103);
await ctl('publish');await until(()=>hints.length===1,'first hint');assert.equal(hints[0],String(BigInt(initialCursor)+1n));
await ctl('disconnect');await until(()=>client.state==='connecting','transient disconnect');await ctl('publish');await until(()=>subscriptions.length===2&&hints.length===2&&catches===2,'recovered hint and HTTP catchup');assert.equal(subscriptions[1].recovered,true);
await ctl('disconnect');await until(()=>client.state==='connecting','second disconnect');await ctl('publish');await ctl('drop');await ctl('publish');await until(()=>subscriptions.length===3&&catches===3,'unrecovered subscription and catchup');assert.equal(subscriptions[2].recovered,false);
await sleep(2600);let stats=await ctl('stats');assert(stats.refresh_js>=1);
await ctl('revoke');await until(()=>client.state==='disconnected','revoked connection stops');const before=events.filter(e=>e.type==='connected').length;await sleep(1200);assert.equal(events.filter(e=>e.type==='connected').length,before);
client.disconnect();
console.log(JSON.stringify({result:'passed',client:'centrifuge-js@5.8.0',node:process.version,auth:'cookie + exact Origin via Node WebSocket headers; browser not executed',badOrigin,foreignChannelCode:denied,publishDenied:103,subscriptions,hints,catches,serverRefreshCount:stats.refresh_js,events},null,2));
