package postgres

import (
 "context"
 "errors"
 "strings"
 "testing"
 "time"
 "github.com/jackc/pgx/v5"
 "github.com/jackc/pgx/v5/pgconn"
 "github.com/jackc/pgx/v5/pgxpool"
)

type cleanupTrace struct{t *testing.T}
type cleanupTraceKey struct{}
type cleanupQuery struct{sql string; started time.Time}
func (tr cleanupTrace) TraceQueryStart(ctx context.Context,_ *pgx.Conn,data pgx.TraceQueryStartData)context.Context{return context.WithValue(ctx,cleanupTraceKey{},cleanupQuery{data.SQL,time.Now()})}
func (tr cleanupTrace) TraceQueryEnd(ctx context.Context,_ *pgx.Conn,data pgx.TraceQueryEndData){q:=ctx.Value(cleanupTraceKey{}).(cleanupQuery); elapsed:=time.Since(q.started); if strings.HasPrefix(q.sql,"DELETE")||data.Err!=nil||elapsed>100*time.Millisecond {code:="";var pgerr *pgconn.PgError;if errors.As(data.Err,&pgerr){code=pgerr.Code};tr.t.Logf("TRACE duration_ms=%.3f command=%q SQLSTATE=%q error_type=%T sql=%q",float64(elapsed.Microseconds())/1000,data.CommandTag.String(),code,data.Err,q.sql)}}
func tracedIntegrationDB(t *testing.T)*Database{t.Helper();db:=integrationDB(t);cfg:=db.pool.Config();cfg.ConnConfig.Tracer=cleanupTrace{t};pool,err:=pgxpool.NewWithConfig(t.Context(),cfg);if err!=nil{t.Fatal(err)};old:=db.pool;db.pool=pool;old.Close();t.Cleanup(pool.Close);return db}
